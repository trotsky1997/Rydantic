// Allocation-only instrumentation: no timers. Synthetic inputs only.
use std::alloc::{GlobalAlloc, Layout, System};
use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering};
use std::{hint::black_box, io::{self, Read}};
use rydantic::{json, Model, Value};
struct Counter;
static ENABLED: AtomicBool = AtomicBool::new(false);
static CALLS: AtomicUsize = AtomicUsize::new(0);
static BYTES: AtomicUsize = AtomicUsize::new(0);
unsafe impl GlobalAlloc for Counter {
    unsafe fn alloc(&self, l: Layout) -> *mut u8 {
        if ENABLED.load(Ordering::Relaxed) { CALLS.fetch_add(1, Ordering::Relaxed); BYTES.fetch_add(l.size(), Ordering::Relaxed); }
        System.alloc(l)
    }
    unsafe fn alloc_zeroed(&self, l: Layout) -> *mut u8 {
        if ENABLED.load(Ordering::Relaxed) { CALLS.fetch_add(1, Ordering::Relaxed); BYTES.fetch_add(l.size(), Ordering::Relaxed); }
        System.alloc_zeroed(l)
    }
    unsafe fn dealloc(&self, p: *mut u8, l: Layout) { System.dealloc(p, l) }
    unsafe fn realloc(&self, p: *mut u8, l: Layout, n: usize) -> *mut u8 {
        if ENABLED.load(Ordering::Relaxed) { CALLS.fetch_add(1, Ordering::Relaxed); BYTES.fetch_add(n, Ordering::Relaxed); }
        System.realloc(p, l, n)
    }
}
#[global_allocator] static ALLOC: Counter = Counter;
fn count<T>(f: impl FnOnce()->T) -> (T, usize, usize) {
    CALLS.store(0,Ordering::Relaxed); BYTES.store(0,Ordering::Relaxed); ENABLED.store(true,Ordering::Relaxed);
    let value = black_box(f());
    ENABLED.store(false,Ordering::Relaxed);
    (value, CALLS.load(Ordering::Relaxed), BYTES.load(Ordering::Relaxed))
}

fn outcome<M: Model>(result: &Result<M, rydantic::ValidationErrors>) -> Value {
    match result {
        Ok(model) => json!({"accepted":true,"output":model.dump().unwrap()}),
        Err(errors) => json!({"accepted":false,"errors":serde_json::to_string(errors).unwrap(),"display":errors.to_string()}),
    }
}
fn parse<M: Model>(input: &str, mode: &str) -> Value {
    assert!(matches!(mode, "json" | "value_copy"));
    let value = if mode == "value_copy" { Some(serde_json::from_str::<Value>(input).unwrap()) } else { None };
    let operation = || match &value {
        Some(value) => M::parse_value(black_box(value).clone()),
        None => M::parse_json(black_box(input)),
    };
    let (result, calls, bytes) = count(operation);
    let expected = outcome(&result);
    drop(result);
    let cold = json!({"calls":calls,"requested_bytes":bytes});
    drop(black_box(operation())); // Initialize parse rule caches before warmed counts.
    let mut warm = Vec::new();
    for _ in 0..5 {
        let (result, calls, bytes) = count(operation);
        assert_eq!(outcome(&result), expected);
        drop(result);
        warm.push(json!({"calls":calls,"requested_bytes":bytes}));
    }
    assert!(warm.iter().all(|row| row == &warm[0]));
    json!({"cold":cold,"warm":warm,"outcome":expected})
}
fn main() {
    let mut input = String::new(); io::stdin().read_to_string(&mut input).unwrap();
    let request: Value = serde_json::from_str(&input).unwrap();
    let text = request["input"].as_str().unwrap();
    let mode = request["mode"].as_str().unwrap();
    let result = match request["schema"].as_str().unwrap() {
        "flat" => parse::<Flat>(text,mode),
        "nested" => parse::<Nested>(text,mode),
        "collection" => parse::<Collection>(text,mode),
        "empty" => parse::<Empty>(text,mode),
        "ignore" => parse::<Ignore>(text,mode),
        "flatten" => parse::<Flatten>(text,mode),
        "extra" => parse::<Extra>(text,mode),
        "tagged" => parse::<Tagged>(text,mode),
        "wide" => parse::<Wide>(text,mode),
        _ => panic!("unknown schema"),
    };
    println!("{}",json!({"result":result,"slot_bytes":std::mem::size_of::<Option<&rydantic::RawValue>>()}));
}
